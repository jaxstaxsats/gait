function calcPace() {

  const km = parseFloat(document.getElementById('log-km').value);

  const timeStr = document.getElementById('log-time').value;

  if (!km || !timeStr) return;

  const parts = timeStr.split(':');

  if (parts.length !== 2) return;

  const totalMins = parseInt(parts[0]) + parseInt(parts[1]) / 60;

  if (!totalMins || km === 0) return;

  const paceMin = Math.floor(totalMins / km);

  const paceSec = Math.round(((totalMins / km) - paceMin) * 60);

  document.getElementById('log-pace').value = `${paceMin}:${paceSec.toString().padStart(2, '0')}`;

}


function selectPain(val) {

  selectedPain = val;

  });

}

        messages: [{ role: 'user', content: prompt }]

      })

    });

    const data = await res.json();

    const text = data.content?.[0]?.text;

    if (text) {

      todayLog.aiResponse = text;

      logs.push(todayLog);

      localStorage.setItem('gait_logs', JSON.stringify(logs));

      bodyEl.innerHTML = formatAIResponse(text);

      updateStats();

    }

  } catch (e) {

    bodyEl.innerHTML = '<p style="color:var(--red)">Something went wrong. Please try again.</p>';

  }


  btn.disabled = false;

  btn.textContent = 'Get today\'s coaching →';

  responseEl.scrollIntoView({ behavior: 'smooth', block: 'start' });

}


function formatAIResponse(text) {

  return text

    .replace(/### (.+)/g, '<h3>$1</h3>')

    .replace(/## (.+)/g, '<h3>$1</h3>')

    .replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>')

    .replace(/^- (.+)/gm, '<li>$1</li>')

    .replace(/\n\n/g, '</p><p>')

    .replace(/^(?!<[hl])/gm, '')

    .split('\n').map(line => {

      if (line.startsWith('<h') || line.startsWith('<li') || line === '') return line;

      return `<p>${line}</p>`;

    }).join('');

}

function updateStats() {

  const runs = logs.filter(l => !l.isRestDay);

  const totalKm = runs.reduce((s, l) => s + parseFloat(l.km || 0), 0);

  document.getElementById('stat-runs').textContent = runs.length;

  document.getElementById('stat-km').textContent = totalKm.toFixed(1);

  // Streak

  let streak = 0;

  const today = new Date().toDateString();

  for (let i = logs.length - 1; i >= 0; i--) {

    const d = new Date(logs[i].date);

    const expected = new Date();

    expected.setDate(expected.getDate() - streak);

    if (d.toDateString() === expected.toDateString()) streak++;

    else break;

  }

  document.getElementById('stat-streak').textContent = streak;

}


function parseMarkdown(text) {

  const lines = text.split('\n');

  const sections = [];

  let current = null;

  for (const line of lines) {

    if (line.startsWith('## ')) {

      if (current) sections.push(current);

      current = { heading: line.replace('## ', ''), content: '' };

    } else if (current) {

      if (line.startsWith('- ') || line.startsWith('• ')) {

        current.content += `<ul><li>${line.replace(/^[-•]\s/, '')}</li></ul>`;

      } else if (line.trim()) {

        current.content += `<p>${line}</p>`;

      }

    }

  }

  if (current) sections.push(current);

  return sections;

}

}
